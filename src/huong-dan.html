app.get("/api/books", (req, res) => {
  const q = req.query.q; 
  if (q) {
    const searchTerm = `%${q}%`;
    const rows = db.prepare("SELECT * FROM books WHERE title LIKE ? OR authors LIKE ?").all(searchTerm, searchTerm);

app.post("/api/books", (req, res) => {
  const { title, authors, thumbnail, shelf } = req.body;
  if (!title) return res.status(400).json({ error: "Tên sách không được để trống!" });

  // [THUẬT TOÁN TẠO ID]: Tự sinh ID ngẫu nhiên.
  const id = crypto.randomUUID();
  const authorsAsString = JSON.stringify(authors);

  try {
    db.prepare(`INSERT INTO books (id, title, authors, thumbnail, shelf) VALUES (?, ?, ?, ?, ?)`)
      .run(id, title, authorsAsString, thumbnail || "", shelf || "none");
    res.json({ success: true, book: { id, title, authors, imageLinks: { thumbnail }, shelf } });
  } catch (error) {
    res.status(500).json({ error: "Lỗi khi thêm sách." });
  }
});

import type { Book, Shelf } from "./types";

export const getBooks = async (query?: string): Promise<Book[]> => {
  const url = query ? `/api/books?q=${encodeURIComponent(query)}` : `/api/books`;
  const res = await fetch(url);
  const data = await res.json();
  return data.books;
};

export const updateBookShelf = async (id: string, shelf: Shelf) => {
  await fetch("/api/books", {
    method: "PUT", 
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ id, shelf }), 
  });
};

export const addNewBook = async (book: Omit<Book, "id">) => {
  const res = await fetch("/api/books", {
    method: "POST", 
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(book),
  });
  return res.json();
};

export const updateFullBook = async (id: string, bookData: any) => {
  const res = await fetch(`/api/books/${id}`, {
    method: "PUT", 
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(bookData),
  });
  return res.json();
};

export const deleteBook = async (id: string) => {
  const res = await fetch(`/api/books/${id}`, {
    method: "DELETE", 
  });
  return res.json();
};

4 Bộ não trung tâm (App.tsx)

import { useEffect, useState } from "react";
import { BrowserRouter, Routes, Route } from "react-router-dom";
import MainPage from "./components/MainPage";
import SearchResultsPage from "./components/SearchResultsPage";
import Navbar from "./components/Navbar";
import AdminPage from "./components/AdminPage";
import { getBooks, updateBookShelf, addNewBook, updateFullBook, deleteBook } from "./api";
import type { Book, Shelf } from "./types";

function App() {
  const [books, setBooks] = useState<Book[]>([]);

  const loadBooks = async () => {
    const data = await getBooks();
    setBooks(data);
  };

  useEffect(() => {
    loadBooks();
  }, []);

  // Hàm đổi kệ (Dành cho User) - OPTIMISTIC UI UPDATE
  const handleShelfChange = async (id: string, newShelf: Shelf) => {
    setBooks((prev) => prev.map((b) => (b.id === id ? { ...b, shelf: newShelf } : b)));
    try {
      await updateBookShelf(id, newShelf);
    } catch (error) {
      loadBooks(); 
    }
  };

  // Hàm thêm sách (Dành cho Admin)
  const handleAddBook = async (newBook: any) => {
    const res = await addNewBook(newBook);
    if (res.success) {
      // [THUẬT TOÁN SPREAD (...)]: Tạo mảng mới gồm sách cũ + sách mới.
      setBooks((prev) => [...prev, res.book]);
    }
  };

  // Hàm sửa sách (Dành cho Admin)
  const handleUpdateBook = async (id: string, bookData: any) => {
    await updateFullBook(id, bookData);
    loadBooks(); 
  };

  // Hàm xóa sách (Dành cho Admin)
  const handleDeleteBook = async (id: string) => {
    // [THUẬT TOÁN FILTER]: Lọc bỏ cuốn sách có id vừa xóa khỏi mảng giao diện.
    setBooks((prev) => prev.filter((b) => b.id !== id));
    await deleteBook(id); 
  };

  return (
    <BrowserRouter>
      <Navbar />
      <Routes>
        <Route path="/" element={<MainPage books={books} onShelfChange={handleShelfChange} />} />
        <Route path="/search" element={<SearchResultsPage onShelfChange={handleShelfChange} />} />
        <Route path="/admin" element={
          <AdminPage 
            books={books} 
            onAddBook={handleAddBook} 
            onUpdateBook={handleUpdateBook} 
            onDeleteBook={handleDeleteBook} 
          />
        } />
      </Routes>
    </BrowserRouter>
  );
}
export default App;

import { useState, useEffect } from "react";
import { useNavigate } from "react-router-dom";
import { getBooks } from "../api";

export default function SearchBar() {
  const navigate = useNavigate();
  const [query, setQuery] = useState("");
  const [showSuggestions, setShowSuggestions] = useState(false);
  const [suggestions, setSuggestions] = useState<string[]>([]);

  // THUẬT TOÁN DEBOUNCE
  useEffect(() => {
    if (query.trim().length < 2) { setSuggestions([]); return; }

    const timer = setTimeout(() => {
      getBooks(query).then((books) => {
        const titles = books.map((b) => b.title);
        const authors = books.map((b) => b.authors).flat().filter(Boolean) as string[];
        const unique = [...new Set([...titles, ...authors])];
        setSuggestions(unique.slice(0, 6));
      });
    }, 300);

    return () => clearTimeout(timer);
  }, [query]);

  const goToResults = (keyword: string) => {
    const trimmed = keyword.trim();
    setShowSuggestions(false);
    navigate(`/search?q=${encodeURIComponent(trimmed)}`);
  };

  return (
    <form onSubmit={(e) => { e.preventDefault(); goToResults(query); }} style={{ width: "100%", maxWidth: "600px" }}>
      <div style={{ position: "relative" }}>
        <input type="text" placeholder="Tìm kiếm..." value={query} onChange={(e) => { setQuery(e.target.value); setShowSuggestions(true); }} onBlur={() => setTimeout(() => setShowSuggestions(false), 200)} style={{ width: "100%", padding: "12px", borderRadius: "99px", border: "1px solid #ccc" }} />
        
        {showSuggestions && suggestions.length > 0 && (
          <ul style={{ position: "absolute", top: "120%", background: "white", width: "100%", zIndex: 10, borderRadius: "8px", boxShadow: "0 10px 15px rgba(0,0,0,0.1)" }}>
            {suggestions.map((suggestion, index) => (
              <li key={index} onMouseDown={(e) => { e.preventDefault(); setQuery(suggestion); goToResults(suggestion); }} style={{ padding: "10px", cursor: "pointer" }}>
                {suggestion}
              </li>
            ))}
          </ul>
        )}
      </div>
    </form>
  );
}

import { useState, useEffect, useRef } from "react";
import type { Shelf } from "../types";

export default function ShelfSelector({ currentShelf, onSelect }: { currentShelf: Shelf, onSelect: (s: Shelf) => void }) {
  const [isOpen, setIsOpen] = useState(false);
  const dropdownRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    function handleClickOutside(event: MouseEvent) {
      if (dropdownRef.current && !dropdownRef.current.contains(event.target as Node)) setIsOpen(false);
    }
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  const labels: Record<Shelf, string> = { currentlyReading: "📖 Đang đọc", wantToRead: "⏳ Sẽ đọc", read: "✅ Đã đọc", none: "❌ Bỏ chọn" };

  return (
    <div ref={dropdownRef} style={{ position: "relative", width: "100%" }}>
      <button onClick={() => setIsOpen(!isOpen)} style={{ width: "100%", padding: "8px", cursor: "pointer", borderRadius: "6px" }}>{labels[currentShelf]}</button>
      {isOpen && (
        <div style={{ position: "absolute", bottom: "110%", left: 0, width: "100%", background: "white", zIndex: 50, boxShadow: "0 -4px 10px rgba(0,0,0,0.1)" }}>
          {(Object.keys(labels) as Shelf[]).map((shelf) => (
            <div key={shelf} onClick={() => { onSelect(shelf); setIsOpen(false); }} style={{ padding: "8px 12px", cursor: "pointer", backgroundColor: currentShelf === shelf ? "#eff6ff" : "white" }}>{labels[shelf]}</div>
          ))}
        </div>
      )}
    </div>
  );
}

        const base64String = reader.result as string;
        setThumbnail(base64String);
        setImagePreview(base64String);
      };
      reader.readAsDataURL(file);
    }
  };

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    const authorsArray = authors.split(",").map(a => a.trim()).filter(Boolean);
    onSave({ title, authors: authorsArray, thumbnail, shelf });
  };

  return (
    <form onSubmit={handleSubmit} style={{ background: "#f8fafc", padding: "20px", borderRadius: "8px", border: "1px solid #cbd5e1", margin: "10px" }}>
      <h4 style={{ marginBottom: "15px" }}>{initialData ? "✏️ Sửa Sách" : "➕ Thêm Sách Mới"}</h4>
      <div style={{ display: "grid", gap: "15px", gridTemplateColumns: "1fr 1fr" }}>
        <div style={{ display: "flex", flexDirection: "column", gap: "10px" }}>
          <div><label style={{ fontSize: "13px", fontWeight: "bold" }}>Tên sách</label><input required value={title} onChange={e => setTitle(e.target.value)} style={{ width: "100%", padding: "8px" }} /></div>
          <div><label style={{ fontSize: "13px", fontWeight: "bold" }}>Tác giả</label><input value={authors} onChange={e => setAuthors(e.target.value)} style={{ width: "100%", padding: "8px" }} /></div>
          <div>
            <label style={{ fontSize: "13px", fontWeight: "bold" }}>Kệ sách</label>
            <select value={shelf} onChange={e => setShelf(e.target.value as Shelf)} style={{ width: "100%", padding: "8px" }}>
              <option value="currentlyReading">Đang đọc</option><option value="wantToRead">Sẽ đọc</option><option value="read">Đã đọc</option><option value="none">Bỏ chọn</option>
            </select>
          </div>
        </div>
        <div style={{ display: "flex", flexDirection: "column", gap: "10px" }}>
          <div>
            <label style={{ fontSize: "13px", fontWeight: "bold", display: "block", marginBottom: "5px" }}>Ảnh bìa (Local hoặc URL)</label>
            <input type="file" accept="image/*" onChange={handleImageUpload} style={{ fontSize: "13px", marginBottom: "10px", display: "block" }} />
            <input placeholder="Hoặc dán link ảnh..." value={thumbnail.startsWith('data:') ? '' : thumbnail} onChange={e => { setThumbnail(e.target.value); setImagePreview(e.target.value); }} style={{ width: "100%", padding: "8px" }} />
          </div>
          {imagePreview && (
            <div><span style={{ fontSize: "12px", color: "#64748b" }}>Xem trước:</span><img src={imagePreview} alt="Preview" style={{ display: "block", width: "80px", height: "120px", objectFit: "cover", marginTop: "5px", borderRadius: "4px" }} /></div>
          )}
        </div>
      </div>
      <div style={{ marginTop: "15px", display: "flex", gap: "10px" }}>
        <button type="submit" style={{ padding: "8px 16px", background: "#2563eb", color: "white", border: "none", borderRadius: "4px", cursor: "pointer" }}>Lưu</button>
        <button type="button" onClick={onCancel} style={{ padding: "8px 16px", background: "#cbd5e1", border: "none", borderRadius: "4px", cursor: "pointer" }}>Hủy</button>
      </div>
    </form>
  );
}

import React, { useState } from "react";
import AdminBookForm from "./AdminBookForm";
import type { Book } from "../types";

interface AdminPageProps {
  books: Book[];
  onAddBook: (book: any) => Promise<void>;
  onUpdateBook: (id: string, book: any) => Promise<void>;
  onDeleteBook: (id: string) => Promise<void>;
}

export default function AdminPage({ books, onAddBook, onUpdateBook, onDeleteBook }: AdminPageProps) {
  const [isAdding, setIsAdding] = useState(false); 
  const [editingBook, setEditingBook] = useState<Book | null>(null); 

  const handleSave = async (data: any) => {
    if (editingBook) {
      await onUpdateBook(editingBook.id, data);
      setEditingBook(null); 
    } else {
      await onAddBook(data);
      setIsAdding(false); 
    }
  };

  const handleDeleteClick = (id: string, title: string) => {
    if (window.confirm(`Bạn có chắc chắn muốn xóa vĩnh viễn cuốn sách "${title}"?`)) {
      onDeleteBook(id);
    }
  };

  return (
    <div style={{ padding: "40px", maxWidth: "1200px", margin: "0 auto" }}>
      <div style={{ display: "flex", justifyContent: "space-between", marginBottom: "20px" }}>
        <h1>🛠️ Trang Quản Trị</h1>
        <button onClick={() => { setIsAdding(true); setEditingBook(null); }} style={{ padding: "10px 20px", background: "#10b981", color: "white", borderRadius: "8px", border: "none", cursor: "pointer", fontWeight: "bold" }}>+ Thêm Sách Mới</button>
      </div>

      {isAdding && (
        <div style={{ marginBottom: "20px", border: "2px dashed #10b981", borderRadius: "8px" }}>
          <AdminBookForm initialData={null} onSave={handleSave} onCancel={() => setIsAdding(false)} />
        </div>
      )}

      <div style={{ background: "white", borderRadius: "8px", overflow: "hidden", border: "1px solid #e2e8f0" }}>
        <table style={{ width: "100%", borderCollapse: "collapse", textAlign: "left" }}>
          <thead style={{ background: "#f8fafc", borderBottom: "1px solid #e2e8f0" }}>
            <tr>
              <th style={{ padding: "12px 16px" }}>Ảnh</th><th style={{ padding: "12px 16px" }}>Tên sách</th><th style={{ padding: "12px 16px" }}>Tác giả</th><th style={{ padding: "12px 16px" }}>Kệ</th><th style={{ padding: "12px 16px", textAlign: "right" }}>Hành động</th>
            </tr>
          </thead>
          <tbody>
            {books.map((book) => (
              <React.Fragment key={book.id}>
                <tr style={{ borderBottom: "1px solid #e2e8f0", background: editingBook?.id === book.id ? "#f1f5f9" : "white" }}>
                  <td style={{ padding: "12px 16px" }}><img src={book.imageLinks?.thumbnail} alt="cover" style={{ width: "40px", height: "60px", objectFit: "cover", borderRadius: "4px" }} /></td>
                  <td style={{ padding: "12px 16px", fontWeight: "600" }}>{book.title}</td>
                  <td style={{ padding: "12px 16px", color: "#64748b" }}>{book.authors?.join(", ")}</td>
                  <td style={{ padding: "12px 16px" }}>{book.shelf}</td>
                  <td style={{ padding: "12px 16px", textAlign: "right" }}>
                    <button onClick={() => { setEditingBook(book); setIsAdding(false); }} style={{ marginRight: "8px", padding: "6px 12px", background: "#eab308", color: "white", border: "none", borderRadius: "4px", cursor: "pointer" }}>Sửa</button>
                    <button onClick={() => handleDeleteClick(book.id, book.title)} style={{ padding: "6px 12px", background: "#ef4444", color: "white", border: "none", borderRadius: "4px", cursor: "pointer" }}>Xóa</button>
                  </td>
                </tr>
                {editingBook?.id === book.id && (
                  <tr style={{ background: "#f8fafc", borderBottom: "2px solid #cbd5e1" }}>
                    <td colSpan={5} style={{ padding: 0 }}>
                      <AdminBookForm initialData={editingBook} onSave={handleSave} onCancel={() => setEditingBook(null)} />
                    </td>
                  </tr>
                )}
              </React.Fragment>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
}

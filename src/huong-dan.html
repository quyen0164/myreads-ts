function formatRows(rows) {
  // [LOGIC]: .map() -> Tạo ra một mảng mới dựa trên mảng cũ.
  // Database trả về cột 'authors' dưới dạng chuỗi '["A","B"]'. React cần mảng ["A","B"].
  return rows.map((r) => ({
    id: r.id,
    title: r.title,
    // [KEYWORD]: JSON.parse() -> Biến chuỗi '["A","B"]' ngược lại thành mảng ["A","B"].
    authors: JSON.parse(r.authors), 
    imageLinks: { thumbnail: r.thumbnail }, 
    shelf: r.shelf,
  }));
}

// ==============================================================================

// [GET] /api/books -> Dùng để lấy sách
app.get("/api/books", (req, res) => {
  // [KEYWORD]: req.query -> Lấy các tham số trên URL (Ví dụ: ?q=Clean -> q = "Clean")
  const q = req.query.q; 
  
  if (q) {
    // [LOGIC]: Tìm kiếm tương đối.
    // [KEYWORD]: LIKE '%...%' -> Tìm chuỗi có chứa từ khóa ở bất kỳ đâu.
    // Sách "Clean Code" sẽ khớp với "%Clean%".
    const searchTerm = `%${q}%`;
    
    // [KEYWORD]: .all() -> Lấy TẤT CẢ các dòng thỏa mãn điều kiện.
    const rows = db.prepare("SELECT * FROM books WHERE title LIKE ? OR authors LIKE ?").all(searchTerm, searchTerm);
    return res.json({ books: formatRows(rows) });
  }
  
  // Nếu không có từ khóa (q rỗng), trả về toàn bộ sách
  const rows = db.prepare("SELECT * FROM books").all();
  res.json({ books: formatRows(rows) });
});

// [PUT] /api/books -> Dùng để cập nhật kệ sách
app.put("/api/books", (req, res) => {
  // [KEYWORD]: req.body -> Lấy dữ liệu React gửi lên (VD: { id: "1", shelf: "read" })
  const { id, shelf } = req.body; 
  
  // [KEYWORD]: .run() -> Thực thi lệnh UPDATE (sửa dữ liệu).
  // Tìm cuốn sách có id tương ứng, đổi cột shelf thành kệ mới.
  db.prepare("UPDATE books SET shelf = ? WHERE id = ?").run(shelf, id);
  
  res.json({ success: true });
});

// Chạy server ở cổng 3001
app.listen(3001, () => console.log(`🚀 Backend đang chạy tại http://localhost:3001`));

2. File `src/api.ts`

import type { Book, Shelf } from "./types";

// [NGỮ PHÁP]: async / await -> Xử lý bất đồng bộ. "Chờ lấy xong dữ liệu rồi mới chạy dòng tiếp theo".
export const getBooks = async (query?: string): Promise<Book[]> => {
  // [KEYWORD]: encodeURIComponent() -> Biến dấu cách thành %20 để đưa lên URL an toàn.
  const url = query ? `/api/books?q=${encodeURIComponent(query)}` : `/api/books`;
  const res = await fetch(url);
  const data = await res.json();
  return data.books;
};

export const updateBookShelf = async (id: string, shelf: Shelf) => {
  await fetch("/api/books", {
    // [KEYWORD]: method: "PUT" -> Báo cho Backend biết đây là lệnh "Cập nhật" (Sửa dữ liệu).
    method: "PUT", 
    headers: { "Content-Type": "application/json" },
    // [KEYWORD]: JSON.stringify() -> Đóng gói object thành chuỗi để gửi qua mạng.
    body: JSON.stringify({ id, shelf }), 
  });
};

4 Bộ não trung tâm (App.tsx)

import { useEffect, useState } from "react";
import { BrowserRouter, Routes, Route } from "react-router-dom";
import MainPage from "./components/MainPage";
import SearchResultsPage from "./components/SearchResultsPage";
import Navbar from "./components/Navbar";
import { getBooks, updateBookShelf } from "./api";
import type { Book, Shelf } from "./types";

function App() {
  // [KEYWORD]: useState -> Tạo ra một "bộ nhớ" (State) cho component.
  const [books, setBooks] = useState<Book[]>([]);

  const loadBooks = async () => {
    const data = await getBooks();
    setBooks(data); 
  };

  // [KEYWORD]: useEffect -> Chạy một đoạn code sau khi giao diện đã hiển thị xong.
  // Mảng rỗng [] ở cuối nghĩa là: "Chỉ chạy đúng 1 lần duy nhất khi mở web".
  useEffect(() => {
    loadBooks();
  }, []);

  // ==============================================================================
  // HÀM XỬ LÝ ĐỔI KỆ SÁCH (OPTIMISTIC UI UPDATE)
  // ==============================================================================
  const handleShelfChange = async (id: string, newShelf: Shelf) => {
    // BƯỚC 1: Cập nhật giao diện NGAY LẬP TỨC (Không chờ API trả về để tránh giật lag).
    setBooks((prevBooks) => {
      // [LOGIC]: .map() -> Duyệt qua mảng 'prevBooks' (mảng sách cũ).
      // - Nếu b.id === id (đúng là cuốn sách đang đổi kệ): Trả về một object mới 
      //   có toàn bộ thuộc tính cũ (...b), nhưng đổi thuộc tính shelf thành 'newShelf'.
      return prevBooks.map((b) => 
        b.id === id ? { ...b, shelf: newShelf } : b
      );
    });
    
    // BƯỚC 2: GỬI XUỐNG DATABASE (Chạy ngầm)
    try {
      await updateBookShelf(id, newShelf); 
    } catch (error) {
      console.error("Lỗi khi lưu DB:", error);
      loadBooks(); // Nếu lưu thất bại, tải lại dữ liệu chuẩn
    }
  };

  return (
    <BrowserRouter>
      <Navbar />
      <Routes>
        <Route path="/" element={<MainPage books={books} onShelfChange={handleShelfChange} />} />
        <Route path="/search" element={<SearchResultsPage onShelfChange={handleShelfChange} />} />
      </Routes>
    </BrowserRouter>
  );
}
export default App;

5 Ô Tìm Kiếm (Debounce)

Tạo file src/components/SearchBar.tsx.

import { useState, useEffect } from "react";
import { useNavigate } from "react-router-dom";
import { getBooks } from "../api";

export default function SearchBar() {
  const navigate = useNavigate();
  const [query, setQuery] = useState("");
  const [showSuggestions, setShowSuggestions] = useState(false);
  const [suggestions, setSuggestions] = useState<string[]>([]);

  // ==============================================================================
  // KỸ THUẬT DEBOUNCE (CHỐNG SPAM API)
  // ==============================================================================
  useEffect(() => {
    if (query.trim().length < 2) {
      setSuggestions([]);
      return;
    }

    // [KEYWORD]: setTimeout -> Đặt một bộ đếm ngược. Sau 300ms thì mới chạy code bên trong.
    const timer = setTimeout(() => {
      getBooks(query).then((books) => {
        const titles = books.map((b) => b.title);
        const authors = books.map((b) => b.authors).flat().filter(Boolean) as string[];
        const unique = [...new Set([...titles, ...authors])]; // Xóa trùng lặp
        setSuggestions(unique.slice(0, 6)); 
      });
    }, 300);

    // [KEYWORD]: clearTimeout -> Hủy bộ đếm.
    // Nếu bạn gõ tiếp chữ "a" (trước khi hết 300ms), hàm cleanup này chạy -> Hủy bộ đếm cũ.
    return () => clearTimeout(timer);
  }, [query]);

  const goToResults = (keyword: string) => {
    const trimmed = keyword.trim();
    if (trimmed.length < 2) return;
    setShowSuggestions(false);
    navigate(`/search?q=${encodeURIComponent(trimmed)}`);
  };

  return (
    <form onSubmit={(e) => { e.preventDefault(); goToResults(query); }}>
      <input
        type="text" value={query}
        onChange={(e) => { setQuery(e.target.value); setShowSuggestions(true); }}
        onBlur={() => setTimeout(() => setShowSuggestions(false), 200)}
      />
      {showSuggestions && suggestions.length > 0 && (
        <ul>
          {suggestions.map((suggestion, index) => (
            <li 
              key={index} 
              // [LOGIC]: onMouseDown chạy TRƯỚC onBlur. Dùng onMouseDown để không bị mất sự kiện click.
              onMouseDown={(e) => { e.preventDefault(); setQuery(suggestion); goToResults(suggestion); }} 
            >
              {suggestion}
            </li>
          ))}
        </ul>
      )}
    </form>
  );
}

7 Nút Dropdown Đổi Kệ

Tạo file src/components/ShelfSelector.tsx.

import { useState, useEffect, useRef } from "react";
import type { Shelf } from "../types";

export default function ShelfSelector({ currentShelf, onSelect }: { currentShelf: Shelf, onSelect: (s: Shelf) => void }) {
  const [isOpen, setIsOpen] = useState(false);
  // [KEYWORD]: useRef -> Tạo một "cái móc" để trỏ thẳng vào một thẻ HTML.
  const dropdownRef = useRef<HTMLDivElement>(null);

  // [THUẬT TOÁN]: Click ra ngoài để đóng menu
  useEffect(() => {
    function handleClickOutside(event: MouseEvent) {
      // Nếu dropdownRef tồn tại VÀ vị trí click KHÔNG NẰM TRONG dropdownRef -> Đóng menu
      if (dropdownRef.current && !dropdownRef.current.contains(event.target as Node)) {
        setIsOpen(false);
      }
    }
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  const labels: Record<Shelf, string> = {
    currentlyReading: "📖 Đang đọc",
    wantToRead: "⏳ Sẽ đọc",
    read: "✅ Đã đọc",
    none: "❌ Bỏ chọn",
  };

  return (
    <div ref={dropdownRef} style={{ position: "relative", width: "100%" }}>
      <button onClick={() => setIsOpen(!isOpen)}>
        {labels[currentShelf]}
      </button>
      {isOpen && (
        <div style={{ position: "absolute", bottom: "110%", left: 0, width: "100%" }}>
          {(Object.keys(labels) as Shelf[]).map((shelf) => (
            <div 
              key={shelf} 
              onClick={() => { onSelect(shelf); setIsOpen(false); }} 
            >
              {labels[shelf]}
            </div>
          ))}
        </div>
      )}
    </div>
  );
}

8 Trang Kết Quả & Sửa CSS

1. File `src/components/SearchResultsPage.tsx`

import { useEffect, useState } from "react";
import { useSearchParams } from "react-router-dom";
import { getBooks } from "../api";
import Book from "./Book";
import type { Book as BookType, Shelf } from "../types";

export default function SearchResultsPage({ onShelfChange }: { onShelfChange: (id: string, s: Shelf) => void }) {
  const [searchParams] = useSearchParams();
  // [THUẬT TOÁN]: .normalize("NFC") -> Ép chuẩn Unicode, chống lỗi rớt dấu tiếng Việt.
  const query = (searchParams.get("q") || "").normalize("NFC"); 
  const [results, setResults] = useState<BookType[]>([]);

  useEffect(() => {
    if (query) {
      getBooks(query).then(setResults);
    }
  }, [query]);

  const handleLocalShelfChange = (id: string, newShelf: Shelf) => {
    setResults(prev => prev.map(b => b.id === id ? { ...b, shelf: newShelf } : b));
    onShelfChange(id, newShelf);
  };

  return (
    <div style={{ maxWidth: "1280px", margin: "48px auto", padding: "0 32px" }}>
      <h2>{query ? `Kết quả tìm kiếm cho "${query}"` : "Nhập từ khóa"}</h2>
      {results.length > 0 && (
        <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(180px, 1fr))", gap: "24px" }}>
          {results.map((b) => (
            <Book key={b.id} book={b} onShelfChange={handleLocalShelfChange} />
          ))}
        </div>
      )}
    </div>
  );
}
